;@ map LVLlev

;@ variant k default
;@ strict 0
;@ ins k
;@ outs k
opcode CompLev_k, 0, kkk
  kval, kin,kout xin
  kVal table kval, giLVLlev
  kVal /= 64
  kIn zkr kin
  kres init 0

  if (kIn >= kVal) then
    kres = 1
    goto run
  endif
kres = 0
run:
  zkw kres, kout
endop

;@ variant a
;@ ins a
;@ outs a
opcode CompLev_a, 0, kkk
kval, kin,kout xin
setksmps 1
kVal table kval, giLVLlev
kVal /= 64
aIn zar kin
kIn = k(aIn)
kres init 0

if (kIn >= kVal) then
  kres = 1
  goto run
endif
kres = 0
run:
zaw a(kres), kout
endop
