;@ variant k default
;@ strict 0 0
;@ ins k k
;@ outs k k
opcode ValSw1_k, 0, kkkkk
kval, kin,kctrl,kout1,kout2 xin
  k1 zkr kin1
  kCtrl zkr kctrl
  kCtrl *= 64
  if (kVal != kCtrl) goto default
  zkw k1, kout2
  goto halt
default:
  zkw k1, kout1
halt:
endop

;@ variant a
;@ ins a a
;@ outs a a
opcode ValSw1_a, 0, kkkkk
kval, kin,kctrl,kout1,kout2 xin
  setksmps 1
  a1 zar kin1
  aCtrl zar kctrl
  kCtrl = k(aCtrl)
  kCtrl *= 64
  if (kVal != kCtrl) goto default
  zaw a1, kout2
  goto halt
default:
  zaw a1, kout1
halt:
endop
