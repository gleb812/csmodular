;@ map ENVtim ENVlev
opcode ADSR, 0, kkkkkkkkk
kA, kD, kS, kR, kKB, kIn1, kIn2, kOut1, kOut2 xin

    ; --- mapping tables ---
    kA table kA, giENVtim
    kD table kD, giENVtim
    kS table kS, giENVlev
    kR table kR, giENVtim
    ; ----------------------

ain2 zar kIn2
kin1 zkr kIn1
kTime init 0
kTime0 init 0
kEnv init 0
kstate init 0
kReset init 1

kA /= 100
kD /= 1000
kS /= 127
kR /= 1000

kGate = kin1

	if kKB!= 1 goto NoKB
		kGate += gkGate
		kGate limit kGate, -1, 1
NoKB:
		kTime0 = kTime
		kEnv0 = kEnv
Next:

if kGate <= 0 goto Free
		if kstate = 4 then
			if kReset == 0 goto NoRes
				kEnv = 0
				kTime = 0
				kEnv0 = 0
				kTime0 = 0
				kgoto OutRes
NoRes:
			kTime0 = 0
			kEnv0 = kEnv
			kTime = 0
		endif
OutRes:
		if kstate == 2 goto Decay
		if kstate == 3 goto ESustain

		kstate = 1 ;Attack
		kEnv += 1/(kr*kA)
		kTime += 1
		kEnv limit kEnv, 0, 1

	if kEnv < 1 goto EnvOut
		kstate = 2; Decay
		kTime = 0
		kEnv0 = 1
		kTime0 = 0
Decay:
		kEnv -= (kEnv0 - kS)/(kr*kD-kTime0)
		kTime += 1

	if (kEnv > kS) goto EnvOut
		kstate = 3 ; Sustain
ESustain:
		kEnv = kS
		kgoto EnvOut
Free:
	if kstate == 0 goto EnvOut

	if kstate == 4 goto Release
		kstate = 4
		kEnv0 = kEnv
		kTime0 = 0
		kTime = 0

Release:
		kEnv -= kEnv0/(kr*kR-kTime0)
		kTime += 1

	kEnv limit kEnv, 0, 1
	if kEnv != 0 goto EnvOut
		kstate = 0
		kTime = 0
		kEnv0 = 0
		kTime0 = 0
EnvOut:
aout1 = ain2 * kEnv
kout2 = kEnv
zaw aout1, kOut1
zkw kout2, kOut2
endop
