    ; --- mapping tables ---
    kFreq table kFreq, giOSCfrq
    kVol table kVol, giCLA000
    kMode table kMode, giCLA000
    ; ----------------------

aout1 init 0
iMode[] fillarray 0, 10, 12, 4, 2

        reinit REINIT_OSC
    endif

iRes = iMode[iNDX]

zaw aout1, kOut1
